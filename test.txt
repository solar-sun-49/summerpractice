// q3
read -p "Enter number : " n
fac=1

do
((fac *= i))
done

echo "The factorial of $n is $fac"

// q5
notprime=1
read -p "Enter number: " n
for((i=2;i<=n/2;i++))
do
if((n % i == 0));
then
    notprime=0
    break
fi
done

if((notprime==1)); then
    echo "Prime"
else
    echo "NOT Prime"
fi
